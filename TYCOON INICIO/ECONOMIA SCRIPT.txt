-- ========================================
-- MODULE SCRIPT: ECONOMIA
-- Guarda o armazém de cada base e a carteira de cada jogador
-- ========================================

local Economia = {}

-- dinheiro esperando coleta em cada base: armazens[tycoon] = valor
Economia.armazens = {}

-- dinheiro de cada jogador: carteiras[UserId] = valor
Economia.carteiras = {}

-- pasta que guarda todas as bases
local pastaTycoons = workspace:WaitForChild("Tycoons")

-- devolve a base onde o objeto está (ou nil se estiver fora de uma base)
function Economia.acharTycoon(objeto)
	local atual = objeto
	while atual and atual.Parent ~= pastaTycoons do
		atual = atual.Parent
	end
	return atual
end

-- diz se o jogador é o dono da base
function Economia.ehDono(tycoon, jogador)
	return tycoon ~= nil and tycoon:GetAttribute("Dono") == jogador.UserId
end

-- devolve a base do jogador (ou nil se ele não tiver base)
function Economia.tycoonDoJogador(jogador)
	for _, tycoon in ipairs(pastaTycoons:GetChildren()) do
		if tycoon:GetAttribute("Dono") == jogador.UserId then
			return tycoon
		end
	end
	return nil
end

-- devolve quanto dinheiro está no armazém da base
function Economia.pegarArmazem(tycoon)
	return Economia.armazens[tycoon] or 0
end

-- soma dinheiro no armazém da base
function Economia.adicionarAoArmazem(tycoon, valor)
	Economia.armazens[tycoon] = Economia.pegarArmazem(tycoon) + valor
end

-- zera o armazém da base
function Economia.esvaziarArmazem(tycoon)
	Economia.armazens[tycoon] = 0
end

-- passa o dinheiro do armazém da base para a carteira do jogador
-- devolve quanto foi coletado
function Economia.coletarParaJogador(tycoon, jogador)
	local valor = Economia.pegarArmazem(tycoon)

	if valor > 0 then
		local id = jogador.UserId
		Economia.carteiras[id] = (Economia.carteiras[id] or 0) + valor
		Economia.esvaziarArmazem(tycoon)
	end

	return valor
end

return Economia
