-- ========================================
-- SCRIPT: GERADOR (DROPPER)
-- Cria um bloco a cada X segundos embaixo do gerador
-- Local: dentro da peça Gerador
-- ========================================

local ServerStorage = game:GetService("ServerStorage")

-- a própria peça do gerador
local gerador = script.Parent

-- ========================================
-- CONFIGURAÇÕES DO GERADOR
-- ========================================
local tipoDeBloco = "BlocoComum"  -- nome do molde dentro de ServerStorage.Blocos
local intervaloCriacao = 3        -- segundos entre cada bloco

-- pega o molde do bloco escolhido
local molde = ServerStorage.Blocos:WaitForChild(tipoDeBloco)

-- ========================================
-- FUNÇÃO: cria um bloco novo embaixo do gerador
-- ========================================
local function criarBloco()
	local bloco = molde:Clone() -- faz uma cópia do molde

	-- posiciona logo abaixo do gerador
	local distancia = gerador.Size.Y / 2 + bloco.Size.Y / 2 + 0.5
	bloco.CFrame = gerador.CFrame * CFrame.new(0, -distancia, 0)

	bloco.Anchored = false -- deixa a física agir (o bloco cai)
	bloco.Parent = workspace -- coloca no mundo, e o script Bloco começa a rodar

	-- o servidor controla a física do bloco (evita que ele "trema")
	bloco:SetNetworkOwner(nil)
end

-- ========================================
-- LOOP INFINITO: cria blocos para sempre
-- ========================================
while true do
	task.wait(intervaloCriacao)

	-- só gera se o gerador já foi comprado
	-- (o botão deixa a peça visível ao comprar)
	if gerador.Transparency < 1 then
		criarBloco()
	end
end
