-- ========================================
-- SCRIPT: GERADOR
-- Gera dinheiro automaticamente a cada X segundos
-- e guarda no armazém do módulo Economia
-- ========================================

-- pega o módulo Economia para usar suas funções e variáveis
local Economia = require(workspace.Economia)

-- ========================================
-- CONFIGURAÇÕES DO GERADOR
-- ========================================
local intervaloGeracao = 5 -- segundos entre cada geração de dinheiro
local dinheiroPorCiclo = 10 -- quanto dinheiro é gerado por vez

-- ========================================
-- LOOP INFINITO: roda pra sempre enquanto o jogo estiver ativo
-- ========================================
while true do
    task.wait(intervaloGeracao) -- "pausa" o script por X segundos antes de continuar

    Economia.adicionarAoArmazem(dinheiroPorCiclo) -- chama a função do módulo

    -- toda vez que o ciclo se repete, mostra quanto tem guardado
    print("Novo ciclo! Dinheiro guardado no armazém: " .. Economia.armazem)
end
