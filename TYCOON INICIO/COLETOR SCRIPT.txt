-- ========================================
-- SCRIPT: COLETOR
-- Passa o dinheiro do armazém da base para a carteira do dono
-- Local: Tycoons.TycoonX.Coletor
-- ========================================

local Players = game:GetService("Players")
local ServerScriptService = game:GetService("ServerScriptService")
local Economia = require(ServerScriptService.Economia)

-- a peça do coletor e a base onde ele está
local parte = script.Parent
local tycoon = Economia.acharTycoon(parte)

-- coleta quando o dono toca
parte.Touched:Connect(function(hit)
	local jogador = Players:GetPlayerFromCharacter(hit.Parent)
	if not jogador then return end

	-- só o dono da base coleta
	if not Economia.ehDono(tycoon, jogador) then return end

	local valor = Economia.coletarParaJogador(tycoon, jogador)

	-- só avisa quando coletou alguma coisa
	if valor > 0 then
		local total = Economia.carteiras[jogador.UserId]
		print(jogador.Name .. " coletou " .. valor .. "! Carteira agora: " .. total)
	end
end)
